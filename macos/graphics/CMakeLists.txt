cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(skyrim_macos_graphics LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Prototype host architecture")
  enable_testing()
endif()
if(NOT APPLE)
  message(FATAL_ERROR "This graphics probe targets macOS")
endif()
if(NOT TARGET SDL2::SDL2)
  find_package(SDL2 CONFIG REQUIRED)
endif()
if(NOT TARGET SDL2)
  add_library(SDL2 INTERFACE)
  target_link_libraries(SDL2 INTERFACE SDL2::SDL2)
endif()
# Upstream uses <SDL2/SDL.h>, whereas Homebrew's imported target exposes
# the SDL2 directory for <SDL.h>. Expose its parent without a fixed prefix.
get_target_property(SKYRIM_SDL_INCLUDE_DIRS SDL2::SDL2 INTERFACE_INCLUDE_DIRECTORIES)
foreach(SKYRIM_SDL_INCLUDE_DIR IN LISTS SKYRIM_SDL_INCLUDE_DIRS)
  if(EXISTS "${SKYRIM_SDL_INCLUDE_DIR}/SDL.h")
    get_filename_component(SKYRIM_SDL_INCLUDE_PARENT "${SKYRIM_SDL_INCLUDE_DIR}" DIRECTORY)
    target_include_directories(SDL2 INTERFACE "${SKYRIM_SDL_INCLUDE_PARENT}")
  endif()
endforeach()
set(SKYRIM_MOLTENVK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../deps/MoltenVK/MoltenVK" CACHE PATH "Pinned MoltenVK package root")
find_library(SKYRIM_MOLTENVK_LIBRARY NAMES MoltenVK PATHS "${SKYRIM_MOLTENVK_ROOT}/dynamic/dylib/macOS" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${SKYRIM_MOLTENVK_ROOT}/include/vulkan/vulkan.h")
  message(FATAL_ERROR "Missing MoltenVK headers; fetch the pinned dependency documented in macos/graphics/README.md")
endif()
if(NOT TARGET Vulkan::Vulkan)
  add_library(Vulkan::Vulkan SHARED IMPORTED)
  set_target_properties(Vulkan::Vulkan PROPERTIES IMPORTED_LOCATION "${SKYRIM_MOLTENVK_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SKYRIM_MOLTENVK_ROOT}/include")
endif()
set(DXVK_VULKAN_INCLUDE_DIR "${SKYRIM_MOLTENVK_ROOT}/include")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/dxvk" "${CMAKE_CURRENT_BINARY_DIR}/dxvk")
get_filename_component(MOLTENVK_LIBRARY_DIR "${SKYRIM_MOLTENVK_LIBRARY}" DIRECTORY)
set_target_properties(dxvk_d3d9 PROPERTIES BUILD_RPATH "${MOLTENVK_LIBRARY_DIR}")
add_executable(graphics_smoke graphics_smoke.cpp)
target_link_libraries(graphics_smoke PRIVATE dxvk_d3d9 SDL2)
target_compile_definitions(graphics_smoke PRIVATE DXVK_NATIVE DXVK_WSI_SDL2)
target_compile_definitions(graphics_smoke PRIVATE SKYRIM_MOLTENVK_LIBRARY_PATH="${SKYRIM_MOLTENVK_LIBRARY}")
target_compile_definitions(graphics_smoke PRIVATE SKYRIM_GRAPHICS_CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/probe.conf")
set_target_properties(graphics_smoke PROPERTIES BUILD_RPATH "${MOLTENVK_LIBRARY_DIR}")
# D3D9 hardware shadow compare (point, emulated PCF, texldp, Dref clamp) on
# this GPU; like graphics_smoke it needs a logged-in WindowServer.
add_executable(depth_compare_smoke depth_compare_smoke.cpp)
target_link_libraries(depth_compare_smoke PRIVATE dxvk_d3d9 SDL2)
target_compile_definitions(depth_compare_smoke PRIVATE DXVK_NATIVE DXVK_WSI_SDL2)
target_compile_definitions(depth_compare_smoke PRIVATE SKYRIM_MOLTENVK_LIBRARY_PATH="${SKYRIM_MOLTENVK_LIBRARY}")
target_compile_definitions(depth_compare_smoke PRIVATE SKYRIM_GRAPHICS_CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/probe.conf")
set_target_properties(depth_compare_smoke PROPERTIES BUILD_RPATH "${MOLTENVK_LIBRARY_DIR}")
# D3D9 base vertex draws natively and with the iOS Simulator's base vertex

add_executable(base_vertex_fold_smoke base_vertex_fold_smoke.cpp)

target_compile_definitions(base_vertex_fold_smoke PRIVATE DXVK_NATIVE DXVK_WSI_SDL2)
target_compile_definitions(base_vertex_fold_smoke PRIVATE SKYRIM_MOLTENVK_LIBRARY_PATH="${SKYRIM_MOLTENVK_LIBRARY}")
target_compile_definitions(base_vertex_fold_smoke PRIVATE SKYRIM_GRAPHICS_CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/probe.conf")
set_target_properties(base_vertex_fold_smoke PROPERTIES BUILD_RPATH "${MOLTENVK_LIBRARY_DIR}")
